1) Full region enforcement across the pipeline

Patch `backend/app/edit_strategies.py`:

```python
from __future__ import annotations

import io
from abc import ABC, abstractmethod
from typing import Any

from PIL import Image

from .providers.factory import get_provider
from .ui_detection import patch_ui_regions

RegionInput = list[dict[str, Any]] | Any | None


class EditStrategy(ABC):
    """Provider-independent editing boundary accepting boxes or masks."""

    @abstractmethod
    async def edit(
        self,
        image_bytes: bytes,
        instruction: str,
        regions: RegionInput = None,
    ) -> bytes:
        ...


async def _apply_region_guard(
    image_bytes: bytes,
    edited_bytes: bytes,
    regions: RegionInput,
) -> bytes:
    if not regions:
        return edited_bytes

    try:
        original = Image.open(io.BytesIO(image_bytes)).convert("RGB")
        candidate = Image.open(io.BytesIO(edited_bytes)).convert("RGB")

        if candidate.size != original.size:
            return edited_bytes

        patched = patch_ui_regions(candidate, regions, radius=3)
        buf = io.BytesIO()
        patched.save(buf, format="PNG")
        return buf.getvalue()
    except Exception:
        # Fail closed: if region patching fails, keep the original provider result
        # and let validation decide whether the output is acceptable.
        return edited_bytes


class ProviderEditStrategy(EditStrategy):
    def __init__(self, provider_name: str):
        self.provider_name = provider_name

    async def edit(
        self,
        image_bytes: bytes,
        instruction: str,
        regions: RegionInput = None,
    ) -> bytes:
        provider = get_provider(self.provider_name)
        edited = await provider.edit(image_bytes, instruction, regions=regions)
        return await _apply_region_guard(image_bytes, edited, regions)


class InstagramCleaningStrategy(ProviderEditStrategy):
    PROFILE = "instagram"
    COORDINATE_RULES = {
        "top_navigation_title_banner": (0.00, 0.00, 1.00, 0.12),
        "story_tray_rings": (0.00, 0.12, 1.00, 0.30),
        "post_engagement_overlay": (0.00, 0.70, 1.00, 0.88),
        "bottom_application_menu_dock": (0.00, 0.88, 1.00, 1.00),
    }

    def regions_for_image(self, image: Image.Image) -> list[dict[str, Any]]:
        width, height = image.size
        return [
            {
                "name": name,
                "box": [
                    round(x1 * width),
                    round(y1 * height),
                    round(x2 * width),
                    round(y2 * height),
                ],
                "confidence": 0.50,
                "profile": self.PROFILE,
            }
            for name, (x1, y1, x2, y2) in self.COORDINATE_RULES.items()
        ]


def strategy_for_preset(preset: str | None, provider_name: str) -> EditStrategy:
    if preset == "instagram_screenshot_cleaner":
        return InstagramCleaningStrategy(provider_name)
    return ProviderEditStrategy(provider_name)


def normalize_regions(regions: RegionInput) -> RegionInput:
    if regions is None or isinstance(regions, list):
        return regions
    if getattr(regions, "ndim", None) == 2:
        return regions
    raise TypeError("regions must be a list of boxes or a 2D mask")
```

```python
import base64
import httpx


class QwenLocalProvider:
    def __init__(self, base_url: str, model: str, timeout: int, edit_path: str | None = None):
        self.base_url = base_url.rstrip("/")
        self.model = model
        self.timeout = timeout
        self.edit_path = edit_path or "/v1/images/edits"

    async def edit(self, image: bytes, instruction: str, regions=None) -> bytes:
        if not self.base_url:
            raise RuntimeError("QWEN_BASE_URL is not configured")

        files = {"image": ("input.png", image, "image/png")}
        data = {"model": self.model, "prompt": instruction}

        async with httpx.AsyncClient(timeout=self.timeout) as client:
            response = await client.post(
                self.base_url + "/" + self.edit_path.lstrip("/"),
                files=files,
                data=data,
            )
            response.raise_for_status()
            if response.headers.get("content-type", "").startswith("image/"):
                return response.content
            body = response.json()

        for item in body.get("data", []):
            if item.get("b64_json"):
                return base64.b64decode(item["b64_json"])
            if item.get("url"):
                async with httpx.AsyncClient(timeout=self.timeout) as client:
                    fetched = await client.get(item["url"])
                    fetched.raise_for_status()
                    return fetched.content
        raise RuntimeError("Qwen returned no image")
```

`backend/app/providers/nano_banana.py`

```python
import base64
import httpx


class NanoBananaProvider:
    def __init__(self, endpoint: str, api_key: str, timeout: int):
        self.endpoint = endpoint
        self.key = api_key
        self.timeout = timeout

    async def edit(self, image: bytes, instruction: str, regions=None) -> bytes:
        if not self.endpoint or not self.key:
            raise RuntimeError("Nano Banana is not configured")

        payload = {
            "contents": [
                {
                    "parts": [
                        {"text": instruction},
                        {"inlineData": {"mimeType": "image/png", "data": base64.b64encode(image).decode()}},
                    ]
                }
            ],
            "generationConfig": {"responseModalities": ["TEXT", "IMAGE"]},
        }

        async with httpx.AsyncClient(timeout=self.timeout) as client:
            response = await client.post(self.endpoint + "?key=" + self.key, json=payload)
            response.raise_for_status()
            body = response.json()

        for candidate in body.get("candidates", []):
            for part in candidate.get("content", {}).get("parts", []):
                data = part.get("inlineData", {}).get("data")
                if data:
                    return base64.b64decode(data)

        raise RuntimeError("Nano Banana returned no image")
```

This ensures the pipeline contract is consistent and the `regions` argument is not silently ignored.

Patch `backend/app/main.py` so the API returns richer status and summary fields:

```python
import json
import uuid
import zipfile
from pathlib import Path

from fastapi import BackgroundTasks, FastAPI, File, Form, HTTPException, UploadFile
from fastapi.responses import FileResponse, HTMLResponse
from fastapi.staticfiles import StaticFiles

from .config import settings
from .image_io import read_stream, save_bytes
from .pipeline import process_job
from .state_store import store


app = FastAPI(title=settings.app_name, version="1.0.0")
frontend_dir = Path(__file__).parents[2] / "frontend"
app.mount("/static", StaticFiles(directory=frontend_dir), name="static")

JOBS = {}
ALLOWED_PROVIDERS = {"mock", "qwen_local", "nano_banana"}
ALLOWED_PRESETS = {
    "clean_mode",
    "preserve_scene",
    "keep_sky",
    "keep_edges",
    "soft_rebuild",
    "instagram_screenshot_cleaner",
}
ALLOWED_EXTENSIONS = {"png", "jpg", "jpeg", "webp"}


def _state_path(job_id):
    return Path(settings.storage_root) / job_id / "state.json"


def _job_summary(results):
    summary = {
        "total": len(results),
        "completed": 0,
        "failed": 0,
        "manual_review": 0,
        "stopped": 0,
    }
    for item in results:
        status = item.get("status")
        if status == "completed":
            summary["completed"] += 1
        elif status == "manual_review":
            summary["manual_review"] += 1
        elif status == "failed":
            summary["failed"] += 1
        elif status == "stopped":
            summary["stopped"] += 1
    return summary


def _start_job(job_id, items, provider_name, preset):
    JOBS[job_id] = {"status": "processing", "results": []}
    store.set(job_id, "processing", provider_name, preset, {"results": []})
    _state_path(job_id).parent.mkdir(parents=True, exist_ok=True)
    save_bytes(
        _state_path(job_id),
        json.dumps({
            "job_id": job_id,
            "status": "processing",
            "provider": provider_name,
            "preset": preset,
        }).encode(),
    )

    async def run():
        results, status = await process_job(job_id, items, provider_name, preset)
        JOBS[job_id] = {
            "status": status,
            "results": results,
            "summary": _job_summary(results),
        }
        store.set(job_id, status, provider_name, preset, {"results": results})
        save_bytes(
            _state_path(job_id),
            json.dumps({
                "job_id": job_id,
                "status": status,
                "provider": provider_name,
                "preset": preset,
                "summary": _job_summary(results),
            }).encode(),
        )

    return run


@app.get("/", response_class=HTMLResponse)
def index():
    return (frontend_dir / "index.html").read_text(encoding="utf-8")


@app.post("/set_model_provider")
async def set_model_provider(provider: str = Form(...)):
    if provider not in ALLOWED_PROVIDERS:
        raise HTTPException(400, "unsupported_provider")
    try:
        settings.validate_provider_settings(provider)
    except ValueError as exc:
        raise HTTPException(400, str(exc)) from exc
    settings.default_provider = provider
    return {"status": "ok", "provider": provider}


@app.get("/healthz")
def health():
    return {
        "status": "ok",
        "provider": settings.default_provider,
        "storage_root": str(settings.storage_root),
    }


@app.get("/api/v1/config")
def config():
    return {
        "providers": sorted(ALLOWED_PROVIDERS),
        "default_provider": settings.default_provider,
        "presets": sorted(ALLOWED_PRESETS),
        "platform": "Windows localhost",
    }


@app.post("/api/v1/jobs", status_code=202)
async def create_job(
    background: BackgroundTasks,
    files: list[UploadFile] = File(...),
    provider: str | None = Form(None),
    preset: str | None = Form(None),
    mode: str | None = Form(None),
):
    if len(files) > 100:
        raise HTTPException(413, "too_many_files")

    provider_name = (provider or settings.default_provider).strip().lower()
    if provider_name not in ALLOWED_PROVIDERS:
        raise HTTPException(400, "unsupported_provider")

    try:
        settings.validate_provider_settings(provider_name)
    except ValueError as exc:
        raise HTTPException(400, str(exc)) from exc

    effective_preset = mode or preset
    if effective_preset and effective_preset not in ALLOWED_PRESETS:
        raise HTTPException(400, "unsupported_preset")

    items = []
    for upload in files:
        filename = Path(upload.filename or "upload").name
        if filename.rsplit(".", 1)[-1].lower() not in ALLOWED_EXTENSIONS:
            raise HTTPException(415, "unsupported_format")

        try:
            data = read_stream(upload.file, settings.max_upload_bytes)
        except ValueError as exc:
            raise HTTPException(413, str(exc))

        items.append((filename, data))

    job_id = uuid.uuid4().hex
    store.enqueue(job_id, provider_name, effective_preset)
    background.add_task(_start_job(job_id, items, provider_name, effective_preset))
    return {
        "job_id": job_id,
        "status": "queued",
        "item_count": len(items),
        "provider": provider_name,
        "preset": effective_preset,
    }


@app.post("/api/v1/jobs/{job_id}/rerun", status_code=202)
async def rerun_job(job_id: str, background: BackgroundTasks):
    state_path = _state_path(job_id)
    manifest = Path(settings.storage_root) / job_id / "manifest.json"
    db_state = store.get(job_id)

    if not state_path.exists() or not manifest.exists():
        raise HTTPException(404, "job_not_rerunnable")

    state = json.loads(state_path.read_text(encoding="utf-8"))
    payload = json.loads(manifest.read_text(encoding="utf-8"))
    items = []

    if db_state:
        state["provider"] = db_state.get("provider") or state.get("provider")
        state["preset"] = db_state.get("preset") or state.get("preset")

    for item in payload.get("results", []):
        matches = list((manifest.parent / "original").glob(f"{item['item_id']}_*"))
        if matches:
            items.append((item.get("filename", matches[0].name), matches[0].read_bytes()))

    provider_name, preset = state.get("provider", settings.default_provider), state.get("preset")
    background.add_task(_start_job(job_id, items, provider_name, preset))
    return {
        "job_id": job_id,
        "status": "queued",
        "provider": provider_name,
        "preset": preset,
    }


@app.get("/api/v1/jobs/{job_id}")
def get_job(job_id: str):
    if job_id in JOBS:
        return {"job_id": job_id, **JOBS[job_id]}

    db_state = store.get(job_id)
    manifest = Path(settings.storage_root) / job_id / "manifest.json"
    state = _state_path(job_id)

    if manifest.exists():
        payload = json.loads(manifest.read_text(encoding="utf-8"))
        payload["summary"] = _job_summary(payload.get("results", []))
        return payload

    if db_state:
        return {
            "job_id": job_id,
            "status": db_state["status"],
            "provider": db_state.get("provider"),
            "preset": db_state.get("preset"),
            "results": db_state.get("payload", {}).get("results", []),
            "summary": _job_summary(db_state.get("payload", {}).get("results", [])),
        }

    if state.exists():
        payload = json.loads(state.read_text(encoding="utf-8"))
        payload["summary"] = _job_summary(payload.get("results", []))
        return payload

    raise HTTPException(404, "job_not_found")
```
